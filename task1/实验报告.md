
   - L2正则化项

## 基础实现

```python
class FullyConnectedLayer:
    """
    全连接层（线性层 + 激活函数）
    """
    # 初始化参数 W、b 和反向传播缓存
    def __init__(self, input_dim, output_dim, activation="relu", init_method="auto", seed=None):
        input_dim = int(input_dim)
        output_dim = int(output_dim)
        if seed is not None:
            rng = np.random.default_rng(seed)
            randn = rng.standard_normal
        else:
            randn = np.random.randn

        self.activation = activation.lower()

        # 初始化策略：选择参数初始化缩放系数
        if init_method == "auto":
            if self.activation in ["relu", "leakyrelu"]:
                scale = np.sqrt(2.0 / input_dim)  # He
            elif self.activation in ["sigmoid", "tanh"]:
                scale = np.sqrt(1.0 / input_dim)  # Xavier
            else:
                scale = 0.01  
        elif init_method == "he":
            scale = np.sqrt(2.0 / input_dim)
        elif init_method == "xavier":
            scale = np.sqrt(1.0 / input_dim)
        else:
            scale = 0.01

        self.W = randn((input_dim, output_dim)) * scale
        self.b = np.zeros((1, output_dim))

        # 反向传播缓存
        self.X_cache = None
        self.Z_cache = None

        # 梯度缓存
        self.dW = None
        self.db = None

        # 优化器状态缓存
        # Momentum vW、vb
        self.vW = np.zeros_like(self.W)
        self.vb = np.zeros_like(self.b)
        # Adam 一阶矩（均值）、二阶矩（方差）
        self.mW = np.zeros_like(self.W)
        self.mb = np.zeros_like(self.b)
        self.sW = np.zeros_like(self.W)
        self.sb = np.zeros_like(self.b)

    # 激活函数及其导数
    def _activate(self, Z):
        if self.activation == "relu":
            return np.maximum(0, Z)
        if self.activation == "sigmoid":
            return 1.0 / (1.0 + np.exp(-np.clip(Z, -50, 50)))
        if self.activation == "tanh":
            return np.tanh(Z)
        if self.activation == "leakyrelu":
            return np.where(Z > 0, Z, 0.01 * Z)
        if self.activation == "linear":
            return Z
        raise ValueError(f"不支持的激活函数: {self.activation}")
    
    # 激活函数的导数，用于反向传播计算 dZ。
    def _activation_grad(self, Z):
        if self.activation == "relu":
            return (Z > 0).astype(float)
        if self.activation == "sigmoid":
            A = 1.0 / (1.0 + np.exp(-np.clip(Z, -50, 50)))
            return A * (1.0 - A)
        if self.activation == "tanh":
            A = np.tanh(Z)
            return 1.0 - A * A
        if self.activation == "leakyrelu":
            grad = np.ones_like(Z)
            grad[Z <= 0] = 0.01
            return grad
        if self.activation == "linear":
            return np.ones_like(Z)
        raise ValueError(f"不支持的激活函数: {self.activation}")

    def forward(self, X):
        """
        前向传播:A = activation(XW + b)
        """
        self.X_cache = X
        Z = X @ self.W + self.b  
        self.Z_cache = Z 
        A = self._activate(Z)
        return A

    def backward(self, dA, l2_lambda=0.0):
        """
        反向传播。
        dA: 损失对当前层激活输出 A 的梯度 ∂Loss/∂A  dX: 输入梯度
        """
        X = self.X_cache
        Z = self.Z_cache
        m = X.shape[0]

        dZ = dA * self._activation_grad(Z)

        # 梯度包含 L2 正则项：lambda/m * W
        self.dW = (X.T @ dZ) / m + (l2_lambda / m) * self.W
        self.db = np.sum(dZ, axis=0, keepdims=True) / m
        dX = dZ @ self.W.T
        return dX

    def backward_from_dZ(self, dZ, l2_lambda=0.0):
        """
        外部已直接得到损失对当前层线性输出 dZ( softmax + 交叉熵组合）
        """
        X = self.X_cache
        m = X.shape[0]

        self.dW = (X.T @ dZ) / m + (l2_lambda / m) * self.W
        self.db = np.sum(dZ, axis=0, keepdims=True) / m
        dX = dZ @ self.W.T
        return dX

    def update_parameters(
        self,learning_rate,optimizer="sgd",
        beta=0.9,beta1=0.9,beta2=0.999,
        epsilon=1e-8,t=1,
    ):
        """
        参数更新：支持 SGD / Momentum / Adam。
        """
        opt = optimizer.lower()
        if opt == "sgd":
            self.W -= learning_rate * self.dW
            self.b -= learning_rate * self.db
            return
        if opt == "momentum":
            self.vW = beta * self.vW + (1 - beta) * self.dW
            self.vb = beta * self.vb + (1 - beta) * self.db
            self.W -= learning_rate * self.vW
            self.b -= learning_rate * self.vb
            return
        if opt == "adam":
            # 一阶矩估计
            self.mW = beta1 * self.mW + (1 - beta1) * self.dW
            self.mb = beta1 * self.mb + (1 - beta1) * self.db
            # 二阶矩估计
            self.sW = beta2 * self.sW + (1 - beta2) * (self.dW ** 2)
            self.sb = beta2 * self.sb + (1 - beta2) * (self.db ** 2)
            # 偏差修正
            mW_hat = self.mW / (1 - beta1 ** t)
            mb_hat = self.mb / (1 - beta1 ** t)
            sW_hat = self.sW / (1 - beta2 ** t)
            sb_hat = self.sb / (1 - beta2 ** t)

            self.W -= learning_rate * mW_hat / (np.sqrt(sW_hat) + epsilon)
            self.b -= learning_rate * mb_hat / (np.sqrt(sb_hat) + epsilon)
            return
        raise ValueError(f"不支持的优化器: {optimizer}")
```

## 进阶优化

- 实现动量（Momentum）/  Adam优化器

  ```python
  # ===== Adam 优化器状态 =====
  self.m_W = np.zeros_like(self.W)  # 一阶矩
  self.v_W = np.zeros_like(self.W)  # 二阶矩
  self.m_b = np.zeros_like(self.b)
  self.v_b = np.zeros_like(self.b)
  
  # ===== Adam 更新 =====
  # 一阶矩
  self.m_W = beta1 * self.m_W + (1 - beta1) * dW
  self.m_b = beta1 * self.m_b + (1 - beta1) * db
  # 二阶矩
  self.v_W = beta2 * self.v_W + (1 - beta2) * (dW ** 2)
  self.v_b = beta2 * self.v_b + (1 - beta2) * (db ** 2)
  # 偏置修正
  m_W_hat = self.m_W / (1 - beta1 ** t)
  v_W_hat = self.v_W / (1 - beta2 ** t)
  m_b_hat = self.m_b / (1 - beta1 ** t)
  v_b_hat = self.v_b / (1 - beta2 ** t)
  # 参数更新
  self.W -= learning_rate * m_W_hat / (np.sqrt(v_W_hat) + epsilon)
  self.b -= learning_rate * m_b_hat / (np.sqrt(v_b_hat) + epsilon)
  
  # ===== 动量 优化器状态 =====
  self.vW = np.zeros_like(self.W)
  self.vb = np.zeros_like(self.b)
  # ===== 动量 更新 =====
  self.vW = beta * self.vW + (1 - beta) * self.dW
  self.vb = beta * self.vb + (1 - beta) * self.db
  self.W -= learning_rate * self.vW
  self.b -= learning_rate * self.vb
  ```

- 添加Dropout层

  ```python
  # ===================== Dropout层 =====================
  class DropoutLayer:
      """
      Dropout 层：以概率 dropout_rate 随机将神经元输出置为 0,
      并缩放剩余激活值，保持期望不变。
      """
      def __init__(self, dropout_rate=0.5):
          self.dropout_rate = dropout_rate
          self.mask = None         
          self.cache = None         
  
      def forward(self, X, training=True):
          """
          前向传播：
          - 训练模式：生成掩码，按 dropout_rate 丢弃神经元，并缩放
          - 测试模式：直接返回输入（不进行 dropout)
          """
          if training and self.dropout_rate > 0:
              p = 1.0 - self.dropout_rate
              self.mask = (np.random.rand(*X.shape) < p).astype(float)
              out = X * self.mask / p
          else:
              out = X
          self.cache = X
          return out
  
      def backward(self, dA):
          """
          反向传播：将梯度通过掩码传播，并缩放。
          """
          if self.mask is not None:
              p = 1.0 - self.dropout_rate
              return dA * self.mask / p
          else:
              return dA
  ```

  ```python
  """

          scale = np.sqrt(2.0 / input_dim)  # He
      elif self.activation in ["sigmoid", "tanh"]:
          scale = np.sqrt(1.0 / input_dim)  # Xavier
      else:
          scale = 0.01  # 线性激活用小随机数
  elif init_method == "he":
      scale = np.sqrt(2.0 / input_dim)
  elif init_method == "xavier":
      scale = np.sqrt(1.0 / input_dim)
  else:
      scale = 0.01
  
  # 使用缩放后的随机数初始化权重 W，偏置 b 初始化为零。
  self.W = randn((input_dim, output_dim)) * scale
  self.b = np.zeros((1, output_dim))
  ```

# 结果分析

## 一、Baseline
- **配置**：隐藏层 [256,128]，ReLU，lr=0.01，batch=64，L2=1e-4，SGD，He 初始化  
- **曲线特征**：训练损失从 0.4324 平稳下降至 0.0173，验证损失同步下降至 0.0811，最终验证准确率 **97.62%**。曲线光滑，无剧烈震荡。  
- **分析**：ReLU 激活函数缓解梯度消失，He 初始化保证前向传播时方差稳定；适中的学习率使 SGD 能稳定更新；L2 正则化（1e-4）轻微约束参数，避免过拟合，因此模型在 80 个 epoch 内稳定收敛，达到良好的泛化性能。

## 二、Group1（宽网络 + LeakyReLU + 强 L2）
- **配置**：隐藏层 [512,256]，LeakyReLU，lr=0.005，batch=128，L2=1e-3，SGD，He 初始化  
- **曲线特征**：初始损失高达 0.8542（远高于 Baseline），收敛速度慢，最终验证损失 0.1204，准确率 **96.47%**，远低于其他组。  
- **分析**：  
  1. **网络过宽**：512→256 的宽度远大于 Baseline，参数数量激增，而 MNIST 任务简单，过宽的模型容易过拟合或收敛困难，需要更小的学习率或更强的正则化。  
  2. **学习率偏低**：0.005 相对于 Baseline 的 0.01 较小，导致参数更新步长不足，尤其在宽网络中梯度可能更小，进一步拖慢收敛。  
  3. **正则化过强**：L2 系数 1e-3 比 Baseline 大 10 倍，导致权重衰减过强，模型欠拟合，即使训练 80 个 epoch 后训练损失仍高达 0.0887，而 Baseline 仅 0.0173，说明模型容量未被充分利用。  
  4. **LeakyReLU 的微小负斜率**对缓解神经元死亡有帮助，但并非性能差的主因；主要问题是网络宽度与正则化强度、学习率不匹配。

## 三、Group2（浅层 + Tanh + 大学习率）
- **配置**：隐藏层 [128]，Tanh，lr=0.02，batch=32，L2=0，SGD，He 初始化  
- **曲线特征**：初始损失 0.3436（低于 Baseline），收敛迅速，最终验证损失 0.0797，准确率 **97.58%**，与 Baseline 接近。  
- **分析**：  
  1. **单隐藏层**：网络深度浅，梯度消失风险小，即使使用 Tanh 也能有效训练。  
  2. **学习率较大（0.02）**：适合浅层网络，能快速下降。  
  3. **无正则化（L2=0）**：虽然可能轻微过拟合（验证损失后期略高于训练损失），但对 MNIST 简单任务影响不大。  
  4. **batch_size=32**：较小的 batch 带来更多梯度噪声，但在该配置下反而有助于跳出局部极小，收敛速度较快。  
  5. **Tanh 在浅层表现尚可**：Tanh 在饱和区梯度很小，但单层网络很少进入饱和区，因此性能未受明显影响。

## 四、Advanced（优化）
- **配置**：隐藏层 [256,128]，ReLU，lr=0.001，batch=64，L2=1e-4，Adam，Xavier 初始化，Dropout=0.2  
- **曲线特征**：训练损失极速下降（第 1 个 epoch 仅 0.1189），很快降至接近零（0.0011），验证损失稳定在 0.08~0.12 之间，验证准确率持续攀升至 **98.30%**，测试准确率 **98.34%**。  
- **分析**：  
  1. **Adam 优化器**：自适应学习率，对不同参数采用不同步长，使训练初期快速收敛，后期平稳，对学习率不敏感，0.001 是常用值，效果好。  
  2. **Xavier 初始化**：适合 ReLU 网络，保持前向/反向传播方差稳定，配合 Adam 使训练稳定。  
  3. **Dropout=0.2**：有效防止过拟合，验证损失未随训练损失降低而升高，泛化能力强。  
  4. **与 Baseline 相同的网络结构**，但通过更先进的优化器、初始化和正则化，性能显著提升。

# 超参数敏感性分析

- **网络宽度与深度**：对 MNIST，适度宽度（256→128）和深度（2 层）已足够，过度加宽（512→256）反而需要更精细的正则化与学习率调整，否则容易欠拟合。  
- **激活函数**：ReLU 在深度适中时表现最佳；Tanh 在浅层可用，但深层易梯度消失；LeakyReLU 本身无缺陷，但在 Group1 中性能差主要因其他超参数不当。  
- **学习率与 Batch Size**：小学习率 + 大 Batch Size 组合更利于稳定收敛，而大学习率 + 小 Batch Size 会显著损害泛化性能。
- **正则化**：L2 正则化需与网络容量匹配，过大导致欠拟合；Dropout 是提升泛化的有效手段，尤其在 Advanced 中效果显著。  
- **优化器**：Adam 相比 SGD 收敛更快、对学习率更鲁棒，在相同网络结构下能获得更高准确率。  
- **初始化**：Xavier 与 He 初始化均能保证训练稳定，但配合不同激活函数效果略有差异，Advanced 中 Xavier 与 Adam 结合良好。

# 讨论

## 不同激活函数对梯度消失的影响
参数梯度的计算依赖激活函数导数与上层梯度的连乘，即链式法则：
$$
\frac{\partial L}{\partial W_l} = \frac{\partial L}{\partial A_L} \cdot \prod_{k=l+1}^{L} \left( W_k^T \cdot \sigma'(Z_k) \right) \cdot X_l
$$
若激活函数导数的绝对值持续小于1，连乘后梯度会指数级衰减；若导数过大则可能引发梯度爆炸。

- Sigmoid 函数
  $$
  \sigma(x) = \frac{1}{1+e^{-x}},\quad \sigma'(x) = \sigma(x)\bigl(1-\sigma(x)\bigr)
  $$
  

  导数最大值为 0.25；当 |x|>4 时，导数趋近于 0；函数输出均值为 0.5，非零中心化，会导致上层输入偏向正值，进一步加剧梯度衰减。

  深度网络中，多层 Sigmoid 的导数连乘，梯度几乎归零，出现严重梯度消失。

- Tanh 函数
  $$
  \tanh(x) = \frac{e^{x} - e^{-x}}{e^{x} + e^{-x}},\quad \tanh'(x) = 1 - \tanh^2(x)
  $$
  在 x=0 处导数最大值为 1，且输出零中心化；当 |x|>3 时，导数趋近于 0。

  零中心化避免了上层输入偏向正值的问题，导数最大值 1 也减少了连乘衰减，梯度消失程度降低。但是深层网络中，若每层 x 都偏离 0，导数连乘仍会快速衰减。

- ReLU 函数
  $$
  \operatorname{ReLU}(x) = \max(0, x),\quad \operatorname{ReLU}'(x) = 
  \begin{cases}
  1, & x > 0 \\
  0, & x \le 0
  \end{cases}
  $$
  

  正区间导数恒为 1，无梯度衰减；存在 “死亡 ReLU” 问题：若 x≤0，导数为 0，对应神经元永久无法更新。

  深层网络中，只要神经元输出为正，梯度就能无衰减地反向传播至浅层，几乎解决梯度消失。但是若学习率过大，易导致 ReLU 神经元死亡，反而引发局部梯度消失。

-  Leaky ReLU 函数
  $$
  \operatorname{Leaky\ ReLU}(x) = 
  \begin{cases}
  x, & x > 0 \\
  \alpha x, & x \le 0
  \end{cases},\quad
  \operatorname{Leaky\ ReLU}'(x) = 
  \begin{cases}
  1, & x > 0 \\
  \alpha, & x \le 0
  \end{cases}
  $$
  正区间导数为 1，无衰减；负区间导数 = 0.01，非零，避免了死亡神经元，梯度可继续传播。

  可以解决 ReLU 的死亡问题，缓解梯度消失。

## 正则化方法如何平衡偏差-方差

- L2 正则化（权重衰减）—— 约束参数幅度

  在损失函数中加入权重的 L2 范数惩罚项：
  $$
  L_{\text{total}} = L_{CE} + \frac{\lambda}{2} \sum_l \| W_l \|_2^2
  $$
  反向传播时，权重梯度会增加$\lambda W_l$，使得更新后的权重更接近 0：
  $$
  W \leftarrow W - \eta\left( \nabla_W L_{CE} + \lambda W \right)
  $$
  L2 正则化强制权重幅度减小，避免个别参数因训练数据噪声被过度放大，让模型更关注全局规律而非局部噪声，大幅降低过拟合。

- Dropout—— 随机失活神经元

  训练时对每层神经元随机选择一部分神经元置 0，仅用剩余的神经元前向 / 反向传播，同时将输出缩放，保证期望不变。测试时关闭 Dropout，所有神经元参与计算，无缩放。

  Dropout 相当于 “训练多个子网络”，测试时相当于 “子网络集成”。集成学习的核心作用是降低方差 —— 单个子网络可能过拟合某类噪声，多个子网络的平均预测会抵消噪声影响，让模型更鲁棒。


## 网络深度与模型性能的关系

网络深度与模型性能之间存在明显的边际收益递减规律，且存在对应的性能拐点。在合理区间内提升网络深度，模型可通过多层特征堆叠，实现底层纹理、局部构件到全局语义特征的逐级抽象，增强对复杂非线性数据分布的拟合能力，精度可得到稳步提升。在参数量相当的前提下，适度加深网络结构，相较于单纯拓宽网络宽度，往往具备更高的特征学习效率、更稳定的收敛过程以及更优的泛化性能。

当网络深度超出合理阈值后，盲目堆叠网络层会引发一系列负面影响。例如，反向传播过程中多层链式求导极易出现梯度消失或梯度爆炸问题，导致浅层网络权重无法有效更新，整体模型难以收敛；深层冗余网络层难以拟合恒等映射，会破坏前期习得的有效特征，引发训练精度与测试精度同步下滑的退化现象，该问题并非单纯的过拟合所致。除此之外，深度过大还会导致参数量激增、推理延时大幅升高、显存占用超标，严重制约模型的实际工程部署。

因此，在设计网络架构时，核心在于保留深度特征学习的优势，同时规避深层网络的固有缺陷。残差直连结构能够让深层网络仅需学习特征增量残差，有效缓解网络退化问题，同时打通梯度反向传播路径；批量归一化与层归一化操作，可稳定每一层的输入数据分布，约束梯度波动范围；配合ReLU类激活函数与Dropout正则化手段，能够进一步抑制梯度饱和，降低过拟合风险，保障深层网络的稳定训练与性能发挥。
